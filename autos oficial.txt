--------------------------------------------------------------------------------
-- LÓGICA DE GIRO (FLING MODERADO)
--------------------------------------------------------------------------------

local function stopFling()
    flingActive = false
    if bav then bav:Destroy() end
    flingButton.Text = "Activar Giro"
    flingButton.BackgroundColor3 = Color3.fromRGB(40, 167, 69)
end

local function startFling(vehicle)
    local targetPart = vehicle.PrimaryPart or vehicle:FindFirstChildWhichIsA("BasePart")
    if not targetPart then return end

    flingActive = true
    flingButton.Text = "Desactivar Giro"
    flingButton.BackgroundColor3 = Color3.fromRGB(220, 53, 69)

    -- Aplicar velocidad angular controlada en el eje Y
    bav = Instance.new("BodyAngularVelocity")
    bav.MaxTorque = Vector3.new(0, math.huge, 0) -- Solo aplica fuerza en el eje Y para mantener estabilidad
    bav.AngularVelocity = Vector3.new(0, flingSpeed, 0)
    bav.Parent = targetPart
end

flingButton.MouseButton1Click:Connect(function()
    local character = LocalPlayer.Character
    if character and character:FindFirstChildOfClass("Humanoid") then
        local seat = character.Humanoid.SeatPart
        if seat and seat.Parent then
            local vehicle = seat.Parent
            if flingActive then
                stopFling()
            else
                startFling(vehicle)
            end
        end
    end
end)

--------------------------------------------------------------------------------
-- LÓGICA DE MOVIMIENTO Y VUELO
--------------------------------------------------------------------------------

local function stopFlying()
    flying = false
    if renderConnection then renderConnection:Disconnect() end
    if bv then bv:Destroy() end
    if bg then bg:Destroy() end
    
    toggleButton.Text = "Activar Vuelo"
    toggleButton.BackgroundColor3 = Color3.fromRGB(40, 167, 69)
end

local function startFlying(vehicle)
    if flying then return end
    flying = true

    toggleButton.Text = "Desactivar Vuelo"
    toggleButton.BackgroundColor3 = Color3.fromRGB(220, 53, 69)

    local targetPart = vehicle.PrimaryPart or vehicle:FindFirstChildWhichIsA("BasePart")
    if not targetPart then return end

    bv = Instance.new("BodyVelocity")
    bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bv.Velocity = Vector3.zero
    bv.Parent = targetPart

    bg = Instance.new("BodyGyro")
    bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
    bg.CFrame = targetPart.CFrame
    bg.Parent = targetPart

    renderConnection = RunService.RenderStepped:Connect(function()
        if not flying or not vehicle or not vehicle.Parent then
            stopFlying()
            return
        end

        local moveDir = Vector3.zero
        local camCF = Camera.CFrame

        if keysPressed[Enum.KeyCode.W] then moveDir = moveDir + camCF.LookVector end
        if keysPressed[Enum.KeyCode.S] then moveDir = moveDir - camCF.LookVector end
        if keysPressed[Enum.KeyCode.A] then moveDir = moveDir - camCF.RightVector end
        if keysPressed[Enum.KeyCode.D] then moveDir = moveDir + camCF.RightVector end
        if keysPressed[Enum.KeyCode.E] then moveDir = moveDir + Vector3.new(0, 1, 0) end
        if keysPressed[Enum.KeyCode.Q] then moveDir = moveDir - Vector3.new(0, 1, 0) end

        bv.Velocity = moveDir * flySpeed
        bg.CFrame = camCF
    end)
end

toggleButton.MouseButton1Click:Connect(function()
    if flying then
        stopFlying()
    else
        local character = LocalPlayer.Character
        if character and character:FindFirstChildOfClass("Humanoid") then
            local seat = character.Humanoid.SeatPart
            if seat and seat.Parent then
                local vehicle = seat.Parent
                startFlying(vehicle)
            end
        end
    end
end)

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not gameProcessed then
        keysPressed[input.KeyCode] = true
    end
end)

UserInputService.InputEnded:Connect(function(input)
    keysPressed[input.KeyCode] = nil
end)

--------------------------------------------------------------------------------
-- SISTEMA DE ARRASTRE DEL HUD
--------------------------------------------------------------------------------
local dragging = false
local dragInput, dragStart, startPos

local function update(input)
    local delta = input.Position - dragStart
    mainFrame.Position = UDim2.new(
        startPos.X.Scale, 
        startPos.X.Offset + delta.X, 
        startPos.Y.Scale, 
        startPos.Y.Offset + delta.Y
    )
end

mainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position

        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

mainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        update(input)
    end
end)
